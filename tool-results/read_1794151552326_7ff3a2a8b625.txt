   302→   302→def html_dari_teks(judul, teks):
   303→   303→    """Teks/markdown -> HTML dokumen bersih untuk print-to-PDF."""
   304→   304→    baris = []
   305→   305→    for ln in teks.splitlines():
   306→   306→        s = ln.strip()
   307→   307→        if not s:
   308→   308→            continue
   309→   309→        h = mod_html.escape(s)
   310→   310→        if s.startswith("# "):
   311→   311→            baris.append(f"<h2>{h[2:]}</h2>")
   312→   312→        elif s.startswith("## "):
   313→   313→            baris.append(f"<h3>{h[3:]}</h3>")
   314→   314→        elif s.startswith("### "):
   315→   315→            baris.append(f"<h4>{h[4:]}</h4>")
   316→   316→        elif re.match(r"^[-*] ", s):
   317→   317→            baris.append(f"<li>{h[2:]}</li>")
   318→   318→        elif re.match(r"^\d+\. ", s):
   319→   319→            baris.append(f"<li>{h}</li>")
   320→   320→        else:
   321→   321→            baris.append(f"<p>{h}</p>")
   322→   322→    isi = "\n".join(baris)
   323→   323→    return f"""<!doctype html><html><head><meta charset="utf-8"><title>{mod_html.escape(judul)}</title>
   324→   324→<style>@page{{size:A4;margin:2cm}} body{{font-family:'DejaVu Sans',sans-serif;font-size:10.5pt;line-height:1.55;color:#1a1a1a}}
   325→   325→h2{{border-bottom:2px solid #2c5f8a;color:#2c5f8a;padding-bottom:4px}} li{{margin-left:1.2em}} footer{{margin-top:2em;font-size:8.5pt;color:#666;border-top:1px solid #ccc;padding-top:.5em}}</style>
   326→   326→</head><body>{isi}<footer>Sumber daring resmi — dirender menjadi dokumen PDF pada {HARI}.</footer></body></html>"""

   395→   395→def _abstrak_openalex(doi):
   396→   396→    w = get_json(f"https://api.openalex.org/works/doi:{doi}?mailto=pustaka.sibi.riset@gmail.com")
   397→   397→    if not w:

   418→   418→def dok_lembar(no, sumber_label, meta_d=None, abstrak=None, catatan_khusus=""):
   419→   419→    """Buat DOCX Lembar Rujukan utk entri tertentu."""
   420→   420→    import docx
   421→   421→    from docx.shared import Pt, Cm
   422→   422→    from docx.enum.text import WD_ALIGN_PARAGRAPH
   423→   423→
   424→   424→    key = KEY[no]
   425→   425→    out = DIR_DOCX / f"{no:02d} - {key} - Lembar-Rujukan.docx"
   426→   426→    d = docx.Document()
   427→   427→    for sec in d.sections:
   428→   428→        sec.top_margin, sec.left_margin = Cm(3), Cm(3)
   429→   429→        sec.bottom_margin, sec.right_margin = Cm(3), Cm(3)
   430→   430→    st = d.styles["Normal"]
   431→   431→    st.font.name = "Times New Roman"
   432→   432→    st.font.size = Pt(12)
   433→   433→    st.paragraph_format.line_spacing = 1.5
   434→   434→
   435→   435→    p = d.add_paragraph()
   436→   436→    p.alignment = WD_ALIGN_PARAGRAPH.CENTER
   437→   437→    r = p.add_run("LEMBAR RUJUKAN\nDaftar Pustaka Proposal — Purwarupa Penerjemah Bahasa Isyarat")
   438→   438→    r.bold = True
   439→   439→    r.font.size = Pt(13)
   440→   440→    d.add_paragraph()
   441→   441→
   442→   442→    def bar(label, nilai):
   443→   443→        pr = d.add_paragraph()
   444→   444→        rr = pr.add_run(f"{label}: ")
   445→   445→        rr.bold = True
   446→   446→        pr.add_run(str(nilai or "—"))
   447→   447→
   448→   448→    bar("Nomor rujukan", f"{no:02d}")
   449→   449→    bar("Sitasi (APA)", CITE.get(no, key))
   450→   450→    if meta_d:
   451→   451→        m = meta_d
   452→   452→        if m.get("judul"):
   453→   453→            bar("Judul", m["judul"])
   454→   454→        if m.get("penulis"):
   455→   455→            bar("Penulis", "; ".join([a for a in m["penulis"] if a]) or "—")
   456→   456→        jurnal = (m.get("jurnal") or {}).get("display_name") or ""
   457→   457→        tahun = m.get("tahun") or ""
   458→   458→        if jurnal or tahun:
   459→   459→            bar("Publikasi", f"{jurnal} ({tahun})")
   460→   460→        bar("DOI", m.get("doi") or "—")
   461→   461→        if m.get("terindeks") is not None:
   462→   462→            bar("Sitasi tercatat (OpenAlex)", m["terindeks"])
   463→   463→    if abstrak:
   464→   464→        pr = d.add_paragraph()
   465→   465→        rr = pr.add_run("Abstrak resmi: ")
   466→   466→        rr.bold = True
   467→   467→        pr.add_run(abstrak)
   468→   468→    pr = d.add_paragraph()
   469→   469→    rr = pr.add_run("Keterangan akses: ")
   470→   470→    rr.bold = True
   471→   471→    teks_ket = catatan_khusus or ("Naskah teks lengkap diterbitkan penerbit berbayar (tidak ada "
   472→   472→                                  "salinan terbuka legal daring). Lembar ini memuat metadata "
   473→   473→                                  "terverifikasi (Crossref/OpenAlex) beserta abstrak resmi sehingga "
   474→   474→                                  "rujukan dapat diverifikasi; naskah penuh tersedia via DOI/perpustakaan.")
   475→   475→    pr.add_run(teks_ket)
   476→   476→    bar("Sumber verifikasi", sumber_label)
   477→   477→    bar("Tanggal verifikasi", HARI)
   478→   478→
   479→   479→    d.add_paragraph()
   480→   480→    pr = d.add_paragraph()
   481→   481→    pr.add_run("Tautan resmi: ").bold = True
   482→   482→    pr.add_run(DOI.get(no) or sumber_label)
   483→   483→    out.parent.mkdir(exist_ok=True)
   484→   484→    d.save(str(out))
   485→   485→    return out
